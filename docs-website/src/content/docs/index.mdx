import { getCollection } from 'astro:content';
import { CardGrid, Icon } from '@astrojs/starlight/components';
import { pluginIcons } from '../../utils/plugin-icons';
import { getContributors } from '../../utils/plugins';

Enjoying our free knowledge and plugins? [Consider leaving us a review](https://www.trustpilot.com/review/pinelab.studio).

## Plugins

<CardGrid>
  {(
    await getCollection(
      'docs',
      ({ id }) => id.startsWith('plugin/') && !id.endsWith('/changelog')
    )
  )
    .sort(
      (a, b) =>
        (b.data.pluginMonthlyDownloads ?? 0) -
          (a.data.pluginMonthlyDownloads ?? 0) ||
        a.data.title.localeCompare(b.data.title)
    )
    .map((plugin) => (
      <article
        className="card plugin-card"
        style={{ '--plugin-color': pluginIcons[plugin.id.slice(7)].color }}
      >
        <div className="plugin-card-icon" aria-hidden="true">
          {pluginIcons[plugin.id.slice(7)].logo ? (
            <img
              src={`/icons/${pluginIcons[plugin.id.slice(7)].logo}`}
              alt=""
              loading="lazy"
              width="48"
              height="48"
            />
          ) : (
            <Icon name={pluginIcons[plugin.id.slice(7)].icon} size="2rem" />
          )}
        </div>
        <h3>
          <a href={`/${plugin.id}/`}>{plugin.data.title}</a>
        </h3>
        <p>{plugin.data.description}</p>
      </article>
    ))}
</CardGrid>

## Recipes

Practical recipes and best practices from our engineering documentation, updated from GitHub on every build.

<CardGrid>
  {(await getCollection('docs', ({ id }) => id.startsWith('recipes/')))
    .sort(
      (a, b) =>
        (a.data.sidebar.order ?? 1) - (b.data.sidebar.order ?? 1) ||
        a.data.title.localeCompare(b.data.title)
    )
    .map((recipe) => (
      <article
        className="card plugin-card"
        style={{ '--plugin-color': 'var(--brand)' }}
      >
        <div className="plugin-card-icon" aria-hidden="true">
          <Icon name="open-book" size="2rem" />
        </div>
        <h3>
          <a href={`/${recipe.id}/`}>{recipe.data.title}</a>
        </h3>
        <p>{recipe.data.description}</p>
      </article>
    ))}
</CardGrid>
